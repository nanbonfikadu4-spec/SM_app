from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI(title="SM (Social Media) API", version="0.1.0")


class UserRegister(BaseModel):
    full_name: str
    username: str
    phone_or_email: str


@app.get("/")
def home():
    return {
        "app_name": "SM (Social Media)",
        "status": "online",
        "message": "Baga gara SM (Social Media) nagaan dhuftan!"
    }


@app.post("/register")
def register_user(user: UserRegister):
    return {
        "status": "success",
        "message": f"Fayyadamaan {user.full_name} (@{user.username}) gara SM tti galmaa'eera.",
        "data": user
    }
